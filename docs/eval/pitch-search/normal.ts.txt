import { head, prod, SR, N, rng } from "./lib";
// ふつうの輪ゴムの音: 倍音ごとに弱まる速さが違う・1 回だけ・弱い・ビブラート・はじいた直後に少し高い
type Opt = { rates: number[]; plucks: number[]; gain: number; vib: number; drop: number };
function buf(f0: number, o: Opt, seed: number) {
  const r = rng(seed); const amps = [1, 0.5, 0.25, 0.12]; const ph = o.plucks.map(() => amps.map(() => 6.28 * (r() + 0.5)));
  const frames = [];
  for (let i = 0; i < 50; i++) { const data = new Float32Array(N);
    for (let k = 0; k < N; k++) { const s = (i * 4800 + k) / SR; let v = 0;
      o.plucks.forEach((p, j) => { if (s < p) return; const dt = s - p; const inst = 1 + o.drop * Math.exp(-dt * 8); const phase = 6.2832 * f0 * (dt + (o.drop / 8) * (1 - Math.exp(-dt * 8)) * 0 + (o.vib / (6.2832 * 5)) * Math.sin(6.2832 * 5 * dt)) * 1;
        amps.forEach((a, h) => (v += a * Math.exp(-dt * o.rates[h]) * Math.sin((h + 1) * (phase + 6.2832 * f0 * (o.drop / 8) * (1 - Math.exp(-dt * 8))) + ph[j][h]))); void inst; });
      data[k] = o.gain * v + 0.004 * r(); }
    frames.push({ t: i * 100, data }); }
  return { sampleRate: SR, frames };
}
const rr = rng(31); const freqs = Array.from({ length: 20 }, () => 80 * Math.pow(7.5, rr() + 0.5));
const P4 = [0.6, 1.7, 2.8, 3.9];
const cases: [string, Opt][] = [
  ["same decay", { rates: [3, 3, 3, 3], plucks: P4, gain: 0.2, vib: 0, drop: 0 }],
  ["harmonics faster 2,3,4,5", { rates: [2, 3, 4, 5], plucks: P4, gain: 0.2, vib: 0, drop: 0 }],
  ["harmonics much faster 1,4,6,8", { rates: [1, 4, 6, 8], plucks: P4, gain: 0.2, vib: 0, drop: 0 }],
  ["fundamental faster 4,2,2,2", { rates: [4, 2, 2, 2], plucks: P4, gain: 0.2, vib: 0, drop: 0 }],
  ["single pluck", { rates: [2, 3, 4, 5], plucks: [0.7], gain: 0.2, vib: 0, drop: 0 }],
  ["weak 0.02", { rates: [3, 3, 3, 3], plucks: P4, gain: 0.02, vib: 0, drop: 0 }],
  ["vibrato ±2%", { rates: [3, 3, 3, 3], plucks: P4, gain: 0.2, vib: 0.02, drop: 0 }],
  ["pitch drop 3%", { rates: [3, 3, 3, 3], plucks: P4, gain: 0.2, vib: 0, drop: 0.03 }],
  ["slow decay 0.5", { rates: [0.5, 0.6, 0.7, 0.8], plucks: P4, gain: 0.2, vib: 0, drop: 0 }],
];
for (const [name, o] of cases) { let hH = 0, hP = 0, bad = 0; const why: Record<string, number> = {};
  freqs.forEach((f, n) => { const b = buf(f, o, n + 1);
    const a = head.analyzePitch(b), p = prod.analyzePitch(b); const ca = head.pitchChecks(a, 0.03), cp = prod.pitchChecks(p, 0.03);
    if (!ca.every((c) => c.ok)) { hH++; for (const c of ca) if (!c.ok) why[c.code] = (why[c.code] ?? 0) + 1; } else if (Math.abs(a.f0_hz - f) / f > 0.03) bad++;
    if (!cp.every((c) => c.ok)) hP++; });
  console.log(`${name}: held HEAD ${hH}/20 PROD ${hP}/20 bad ${bad} ${JSON.stringify(why)}`); }
